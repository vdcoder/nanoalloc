foreach(example basic_usage containers)
    add_executable(example_${example} ${example}.cpp)
    target_link_libraries(example_${example} PRIVATE nanoalloc::nanoalloc)
    nanoalloc_apply_warnings(example_${example})
    if(BUILD_TESTING)
        add_test(NAME example.${example} COMMAND example_${example})
        set_tests_properties(example.${example} PROPERTIES LABELS "example")
    endif()
endforeach()
